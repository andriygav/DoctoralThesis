\section{Актуальность, задачи и методы исследований}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Актуальность}
\justifying
\begin{itemize}
    \justifying
    \item \textbf{Рост сложности моделей:} от тысяч до сотен миллиардов параметров, прогнозируемый переход к триллионам.
    \item \textbf{Затратность разработки:} большие языковые модели (LLM) требуют огромных вычислительных, энергетических и финансовых ресурсов.
    \item \textbf{Неэффективность масштабирования:} отсутствие строгой теоретической основы для предсказания поведения моделей делает процесс разработки экономически и энергетически неэффективным.
    \item \textbf{Эмпирические решения:} текущие подходы опираются на корреляции, что приводит к необоснованным и противоречивым результатам.
    \item \textbf{Консервативность классических мер:} VC-размерность и сложность Радемахера завышают оценки для перепараметризованных сетей и не учитывают специфику архитектур.
\end{itemize}

\vspace{0.3cm}
\textbf{Ключевая проблема:} отсутствует единый теоретический аппарат для описания сложности моделей и данных с формальными критериями их соответствия.
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Цель и задачи исследования}
\justifying
\textbf{Цель:} Построение единого теоретического аппарата для оценки сложности моделей глубокого обучения и сложности данных, а также установление формальных критериев соответствия между сложностью модели и сложностью выборки, необходимой для ее обучения.

\vspace{0.3cm}
\textbf{Задачи:}
\begin{enumerate}
    \justifying
    \item[--] введение формальных определений мер сложности моделей и данных в рамках теории мер и установление критерия согласованности модели и выборки;
    \item[--] получение теоретических оценок ландшафтных мер на основе матриц Гессе для полносвязных, сверточных и трансформерных архитектур моделей глубокого обучения; установление связи ландшафтной меры с условной сложностью выборки;
    \item[--] построение методов оценки достаточного объема выборок из простой генеральной совокупности;
    \item[--] построение методов снижения сложности моделей глубокого обучения;
    \item[--] демонстрация практического применения построенного аппарата в прикладных задачах.
\end{enumerate}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Научная новизна}
\justifying
\begin{itemize}
    \justifying
    \item Введены формальные определения меры сложности выборки $\mu_D(D)$ и меры сложности модели $\mu_f(f)$ в рамках теории мер, установлен критерий согласованности $\mu_f(f) \le \mu_D(D)$.
    \item Введена ландшафтная мера сложности модели $\mu_f(f|D) = \mathbb{E}\|\mathbf{H}_i - \mathbb{E}\mathbf{H}_i\|$, определяемая через спектральные свойства матриц Гессе.
    \item Получены строгие теоретические оценки ландшафтной меры на основе спектральных норм матриц Гессе для полносвязных, сверточных и трансформерных архитектур.
    \item Установлена связь между сходимостью ландшафта и достаточным объемом выборки; построены критерии D-, M-, KL- и S-достаточности с доказанными теоремами сходимости в модели линейной регрессии.
    \item Предложены методы снижения сложности: прореживание на основе ковариации градиентов, многодоменная дистилляция и антидистилляция.
    \item Доказаны теоремы о состоятельности LoRA-адаптеров и снижении радемахеровской сложности в многозадачном обучении.
\end{itemize}
\end{frame}

    \item Рост выразительности с глубиной (Хестад, 1987; Бенджио, 2007)

\end{itemize}
\end{column}

\vspace{0.15em}
\textbf{Вывод:} направления развивались независимо, единой теории нет.

\section{Общее введение теории ландшафтной меры}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Генеральная совокупность и мера сложности выборки}
\justifying

\begin{rusdefinition}
    \justifying
    Генеральной совокупностью данных~$\Gamma$ назовем произвольное множество объектов, которые исследуются в рамках той либо иной задачи.
\end{rusdefinition}

\begin{rusdefinition}
    \justifying
    Генеральную совокупность~$\Gamma$ назовем однородной, если все объекты генеральной совокупности порождаются из одного распределения. В противном случае назовем ее гетерогенной; если совокупность образуется~$k$ различными распределениями, будем говорить о~$k$-компонентной совокупности.
\end{rusdefinition}

\vspace{0.1cm}
Рассмотрим кольцо выборок: $\mathfrak{D} = \{D_\Gamma^i\}, \quad D_\Gamma^i \subset \Gamma.$

\begin{rusdefinition}
    \justifying
    Мерой сложности выборки назовем отображение~$\mu_D$ такое, что $\mu_D(D_i):\mathfrak{D}_\Gamma\to\mathbb{R}_+$, удовлетворяющее свойству субаддитивности:
    \[
        \mu_D(D_i\cup D_j) \leq \mu_D(D_i)+\mu_D(D_j),
    \]
    где равенство достигается при условии~$D_i\cap D_j=\emptyset.$
\end{rusdefinition}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Мера сложности модели и примеры мер}
\justifying

\begin{rusdefinition}
\justifying
    Мерой сложности модели~$f$ назовем отображение~$\mu_f(f_i)$:
    \[
        \mu_f(f_i) : \mathfrak{F} \to \mathbb{R}_+.
    \]
\end{rusdefinition}

\vspace{0.2cm}
\textbf{Примеры мер:}
\begin{itemize}
    \item Число параметров модели.
    \item Норма вектора весов $\|\mathbf{w}\|$.
    \item Эффективная емкость, оцениваемая через след матрицы Гессе.
    \item В дальнейшем будет введена \textit{ландшафтная мера} $\mu_f(f|D)$, учитывающая данные.
\end{itemize}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Условная сложность выборки}
\justifying
\small

Мера $\mu_D(D)$ оценивает абсолютную сложность данных, но для конкретной модели важна \textit{относительная} сложность.

\begin{rusdefinition}
    \justifying
    Условная сложность выборки $D$ относительно модели $f$:
    \[
        \mu_D(D|f) = \inf\{\mu_D(D'): D'\subseteq D,\; \mu_f(f) \le \mu_D(D')\}.
    \]
    Это минимальная сложность подвыборки, достаточная для выполнения критерия согласованности $\mu_f(f)\le\mu_D(D')$.
\end{rusdefinition}

\vspace{0.1cm}
\begin{block}{Свойства}
    \begin{itemize}
        \item $\mu_D(D|f) \le \mu_D(D)$
        \item Зависит как от данных, так и от архитектуры модели
        \item Монотонность $\mu_D$ сохраняет критерий согласованности при расширении выборки
    \end{itemize}
\end{block}

\vspace{0.15em}
\key{Ключевая идея.} Условная сложность связывает модель и выборку через критерий согласованности $\mu_f(f)\le\mu_D(D')$.
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Простая генеральная совокупность}
\justifying

\begin{block}{Простая генеральная совокупность}
\justifying
Однородную генеральную совокупность~$\Gamma_C$ назовем простой, если она состоит из объектов одинаковой сложности~$C$: $\forall \gamma\in\Gamma_C\; \mu_D(\{\gamma\}) = C$.

\begin{rustheorem}[\emph{Простая генеральная совокупность}]
    Для простой совокупности мера сложности любой конечной выборки пропорциональна ее размеру:
    \[
        \mu_D(D) = C\cdot|D|.
    \]
\end{rustheorem}

Тогда условная сложность превращается в минимальный размер подвыборки:
\[
    \mu_D(D|f) = C\cdot m^*, \quad m^* = \min\{k: \mu_f(f) \le Ck\}.
\]
$m^*$ задает минимальное количество объектов для обучения модели $f$.
\end{block}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{От общей теории к ландшафтной мере: изменение ландшафта при вариации выборки}
\justifying

\textbf{Шаг 1.} Задана выборка из генеральной совокупности~$\Gamma_C$:
\[
    D = \left\{ (\mathbf{x}_i, \mathbf{y}_i) \right\}, \quad i = 1, \ldots, m, \quad \mathbf{x} \in \mathcal{X}, \ \mathbf{y} \in \mathcal{Y}, \quad D\subset \Gamma_C.
\]

\textbf{Шаг 2.} Эмпирический риск на выборках размера $k$ и $k+1$:
\[
\mathcal{L}_k(\boldsymbol{\theta}) = \frac{1}{k}\sum_{i=1}^k \ell(f_{\boldsymbol{\theta}}(\mathbf{x}_i), \mathbf{y}_i), \quad 
\mathcal{L}_{k+1}(\boldsymbol{\theta}) = \frac{1}{k+1}\sum_{i=1}^{k+1} \ell(f_{\boldsymbol{\theta}}(\mathbf{x}_i), \mathbf{y}_i).
\]

\textbf{Шаг 3.} Разность рисков показывает вклад нового объекта:
\[
\mathcal{L}_{k+1}(\boldsymbol{\theta}) - \mathcal{L}_k(\boldsymbol{\theta}) = 
\frac{1}{k+1}\Bigl(\ell(f_{\boldsymbol{\theta}}(\mathbf{x}_{k+1}), \mathbf{y}_{k+1}) - \mathcal{L}_k(\boldsymbol{\theta})\Bigr).
\]

\vspace{0.15cm}
\textbf{Свойства разности}
\begin{itemize}
    \item При $k\to\infty$ разность стремится к нулю.
    \item Скорость убывания определяется тем, насколько новый объект не похож на уже имеющиеся.
    \item Далее эта скорость оценивается через спектральные свойства матрицы Гессе.
\end{itemize}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Предположение о стационарности точки минимума}
\justifying

\begin{rusassumption}
\justifying
Пусть $\boldsymbol{\theta}^*$~--- локальный минимум обеих функций потерь $\mathcal{L}_k(\boldsymbol{\theta})$ и $\mathcal{L}_{k+1}(\boldsymbol{\theta})$, т.е.
\[
\setlength\abovedisplayskip{0pt}
    \nabla\mathcal{L}_k(\boldsymbol{\theta}^*) = \nabla\mathcal{L}_{k+1}(\boldsymbol{\theta}^*) = \mathbf{0}.
\setlength\belowdisplayskip{0pt}
\]
\end{rusassumption}

В окрестности минимума функции аппроксимируются квадратично:
\[
\setlength\abovedisplayskip{0pt}
\mathcal{L}_k(\boldsymbol{\theta}) \approx \mathcal{L}_k(\boldsymbol{\theta}^*) + \frac{1}{2}(\boldsymbol{\theta} - \boldsymbol{\theta}^*)^\top \mathbf{H}^{(k)}(\boldsymbol{\theta}^*)(\boldsymbol{\theta} - \boldsymbol{\theta}^*).
\setlength\belowdisplayskip{0pt}
\]

Тогда разность потерь принимает вид:
\[
\setlength\abovedisplayskip{0pt}
\begin{aligned}
\mathcal{L}_{k+1}(\boldsymbol{\theta}) - \mathcal{L}_k(\boldsymbol{\theta}) &\approx 
\frac{1}{k+1}\Bigl(\ell(f_{\boldsymbol{\theta}^*}(\mathbf{x}_{k+1}), \mathbf{y}_{k+1}) - \mathcal{L}_k(\boldsymbol{\theta}^*)\Bigr) \\
&\quad + \frac{1}{k+1}(\boldsymbol{\theta} - \boldsymbol{\theta}^*)^\top 
\Bigl(\mathbf{H}_{k+1}(\boldsymbol{\theta}^*) - \frac{1}{k}\sum_{i=1}^{k}\mathbf{H}_i(\boldsymbol{\theta}^*)\Bigr)(\boldsymbol{\theta} - \boldsymbol{\theta}^*).
\end{aligned}
\setlength\belowdisplayskip{0pt}
\]

\begin{columns}[c]
    \begin{column}{0.7\textwidth}
        \begin{rusremark}
            \justifying
            Первое слагаемое ограничено константой, не зависящей от $k$. Второе содержит разность матриц Гессе~--- именно она определяет скорость сходимости ландшафта и используется в основе ландшафтной меры.
        \end{rusremark}
    \end{column}
    \begin{column}{0.3\textwidth}
        \centering
        \includegraphics[width=\textwidth]{thesis/figures/chapter-1/transformer/assumption}
    \end{column}
\end{columns}

\end{frame}

\begin{frame}{Ландшафтная мера сложности модели}
\justifying

\begin{rusdefinition}
\justifying
    Ландшафтной мерой сложности параметрической функции $f$ называется
    \[
        \mu_f(f|D) = \mathbb{E}_{\mathbf{x}_i\in D}\left\| \mathbf{H}_i(\boldsymbol{\theta}^*) - \mathbb{E}_{\mathbf{x}_j\in D}\mathbf{H}_j(\boldsymbol{\theta}^*) \right\|_2,
    \]
    где $\mathbf{H}_i(\boldsymbol{\theta}^*)$~--- матрица Гессе для $i$-го объекта в точке минимума $\boldsymbol{\theta}^*$.
\end{rusdefinition}

Ландшафтная мера характеризует отклонение матриц Гессе по объектам выборки:
\begin{itemize}
    \item Малое отклонение~--- объекты одинаково влияют на ландшафт, модель насытилась данными.
    \item Большое отклонение~--- разные объекты по-разному искривляют ландшафт, данных недостаточно.
\end{itemize}

\begin{block}{Связь с изменением функции потерь}
\[
    \left| \mathcal{L}_{k+1}(\boldsymbol{\theta}) - \mathcal{L}_k(\boldsymbol{\theta}) \right| \leqslant \frac{M_{\ell}}{k+1}+ \frac{\mu_f(f|D)}{k+1}\left\|\boldsymbol{\theta} - \boldsymbol{\theta}^*\right\|_2^2.
\]
\end{block}

\vspace{0.15em}
\key{Ключевой результат.} Ландшафтная мера $\mu_f(f|D)$ задает скорость сходимости разности потерь, связывая сложность модели со сложностью выборки.
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Схема вывода: ландшафтная мера сложности}
\justifying
\small
\begin{enumerate}
    \item Разность рисков через вклад одного объекта:
    \[
        \mathcal{L}_{k+1}-\mathcal{L}_k = \frac{1}{k+1}\left(\ell(f_{\boldsymbol{\theta}}(\mathbf{x}_{k+1}),\mathbf{y}_{k+1})-\mathcal{L}_k\right).
    \]
    \item $\boldsymbol{\theta}^*$~--- минимум обеих функций. Разложение Тейлора до второго порядка:
    \[
        \mathcal{L}_k \approx \mathcal{L}_k(\boldsymbol{\theta}^*) + \frac{1}{2}(\boldsymbol{\theta}-\boldsymbol{\theta}^*)^\top\mathbf{H}^{(k)}(\boldsymbol{\theta}-\boldsymbol{\theta}^*).
    \]
    \item Неравенство треугольника. Первое слагаемое ограничено $\frac{M_\ell}{k+1}$. Второе~--- через норму разности матриц Гессе $\left\|\mathbf{H}_{k+1}-\frac{1}{k}\sum_i\mathbf{H}_i\right\|_2$.
    \item Мажорирование разности Гессе через разброс пообъектных матриц $2\,\mu_f(f|D)$ дает итоговую оценку сходимости.
\end{enumerate}
\end{frame}

\section{Оценка матриц Гессе для нейросетевых моделей}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Декомпозиция матрицы Гессе на G и H компоненты}
\justifying
\small

Матрица Гессе для функции $\mathcal{L}(\boldsymbol{\theta})$ от $n$ параметров:
\[
    \mathbf{H}(\mathcal{L})_{ij} = \frac{\partial^2 \mathcal{L}}{\partial \theta_i \partial \theta_j} = \sum_{k=1}^{m}\frac{\partial^2 \ell(\mathbf{z}_k, \mathbf{y}_k)}{\partial \theta_i \partial \theta_j},
\]

Используя цепное правило, матрица Гессе представима в виде двух слагаемых:
\[
    \mathbf{H}_{i}(\boldsymbol{\theta}) = \underbrace{\nabla_{\boldsymbol{\theta}} \mathbf{z}_i \dfrac{\partial^2 \ell(\mathbf{z}_i, \mathbf{y}_i)}{\partial \mathbf{z}_{i}^2} \nabla_{\boldsymbol{\theta}} \mathbf{z}_i^{\text{T}}}_{\text{G-компонента}} + \underbrace{\sum\limits_{k=1}^{K} \dfrac{\partial \ell(\mathbf{z}_i, \mathbf{y}_i)}{\partial z_{ik}} \nabla^2_{\boldsymbol{\theta}} z_{ik}}_{\text{H-компонента}}.
\]

\begin{ruscorollary}
    \justifying
    Эмпирически установлено, что H-компонента близка к нулю, а особый интерес представляет G-компонента:
\[
    \mathbf{H}_{i}(\boldsymbol{\theta}) \approx \nabla_{\boldsymbol{\theta}} \mathbf{z}_i \dfrac{\partial^2 \ell(\mathbf{z}_i, \mathbf{y}_i)}{\partial \mathbf{z}_{i}^2} \nabla_{\boldsymbol{\theta}} \mathbf{z}_i^{\text{T}}.
\]
\end{ruscorollary}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Оценка матрицы Гессе для полносвязной сети}
\footnotesize
\begin{rustheorem}[\emph{Оценка нормы матрицы Гессе для полносвязных сетей}]
    \justifying
    Для $L$-слойной полносвязной нейронной сети с ReLU, применяемой к задаче классификации на $K$ классов, пусть $\| \mathbf{W}^{(p)} \|_2 \leqslant M_{\mathbf{W}}$ и $\| \mathbf{x}_i \|_2 \leqslant M_{\mathbf{x}}$ для всех слоев $p = 1, \ldots, L$ и объектов $i = 1, \ldots, m$.
    Тогда для любого объекта $i = 1, \ldots, m$ выполняется неравенство
    \[
        \left\| \mathbf{H}_i(\boldsymbol{\theta}) \right\|_2 \leqslant L \sqrt{2} M_{\mathbf{x}}^2 M_{\mathbf{W}}^{2L} + \sqrt{2} \frac{M_{\mathbf{W}}^2 (M_{\mathbf{W}}^{2L} - 1)}{M_{\mathbf{W}}^2 - 1}.
    \]
\end{rustheorem}

\begin{rustheorem}[\emph{Асимптотика нормы матрицы Гессе для полносвязных сетей}]
    \justifying
    Пусть все параметры модели ограничены константой~$M > 0$, то есть $|w_{ij}^{(p)}| \leqslant M$ для всех $i, j = 1, \ldots, h$ и всех слоев $p = 1, \ldots, L$. Тогда при выполнении условий предыдущей теоремы справедлива оценка
    \[
        \left\| \mathbf{H}_i(\boldsymbol{\theta}) \right\|_2 \leqslant L \sqrt{2} M_{\mathbf{x}}^2 (hM)^{2L} + \sqrt{2} \frac{(hM)^2 ((hM)^{2L} - 1)}{(hM)^2 - 1}.
    \]
\end{rustheorem}

\begin{ruscorollary}
    \justifying
    Для однослойной сети ($L=1$) получаем
    \[
        \left\| \mathbf{H}_i(\boldsymbol{\theta}) \right\|_2 \leqslant \sqrt{2} M_{\mathbf{W}}^{2} \left( M_{\mathbf{x}}^2 + 1 \right).
    \]
\end{ruscorollary}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Схема доказательства: оценка нормы Гессе для полносвязной сети}
\justifying
\footnotesize
\begin{itemize}
    \item ReLU-маски и матрицы Якоби:
    \[
        \mathbf{G}^{(p)} = \mathbf{W}^{(L)}\mathbf{D}^{(L-1)}\mathbf{W}^{(L-1)}\cdots\mathbf{D}^{(p)}
    \]
    (произведение весов и диагональных масок активаций).
    \item Якобиан логитов по параметрам слоя: $\nabla_{\boldsymbol{\theta}}\mathbf{z}_i = (\mathbf{G}^{(p)})^\top \otimes \mathbf{x}^{(p)}$ (структура Кронекера).
    \item Факторизация G-компоненты: $\mathbf{H}_i(\boldsymbol{\theta}) \approx \mathbf{F}_i^{\top}\mathbf{A}_i\mathbf{F}_i$ с $\mathbf{A} = \mathrm{diag}(\mathbf{p})-\mathbf{p}\mathbf{p}^{\top}$ (внутренний гессиан softmax).
    \item Спектральная норма: каждое умножение на $\mathbf{W}^{(p)}$ вносит множитель $\|\mathbf{W}^{(p)}\|_2 \leqslant M_{\mathbf{W}}$, откуда $M_{\mathbf{W}}^{2L}$ после $L$ слоев и суммирование геометрической прогрессии.
\end{itemize}

\vspace{0.15em}
\key{Ключевая идея.} Норма Гессе факторизуется через произведение весов слоев, что дает множитель $M_{\mathbf{W}}^{2L}$.
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Факторизация матричных моделей глубокого обучения}
\justifying
\small
Пусть $f_{\boldsymbol{\theta}}(\mathbf{x})$ является суперпозицией линейных операторов:
\begin{equation*}
    f_{\boldsymbol{\theta}}(\mathbf{x}) = \mathbf{T}^{(L+1)}\boldsymbol{\Lambda}^{(L)}\dots \boldsymbol{\Lambda}^{(1)}\mathbf{T}^{(1)}\mathbf{x},
\end{equation*}
где~$\boldsymbol{\theta} = \mathrm{col}(\mathbf{W}^{(L+1)}, \dots, \mathbf{W}^{(1)}).$

\begin{rustheorem}[\emph{Факторизация матрицы Гессе}]
    \justifying
    Пусть функция нейронной сети~$f_{\boldsymbol{\theta}}(\mathbf{x})$ представима в виде суперпозиции линейных операторов, тогда матрица Гессе функции потерь относительно параметров модели представляется в факторизованной форме:
    $\mathbf{H}_O(\boldsymbol{\theta}) = \mathbf{Q}^{T}\mathbf{F}^{T}\mathbf{A}\mathbf{F}\mathbf{Q}$, где $\mathbf{H}_O$ описывает G-компоненту матрицы Гессе.
\end{rustheorem}

\begin{rustheorem}[\emph{Оценка нормы матрицы Гессе для матричных сетей}]
    \justifying
    Пусть функция нейронной сети~$f_{\boldsymbol{\theta}}(\mathbf{x})$ представима в виде суперпозиции линейных операторов.
    Пусть для всех $p$ выполняется $\left\|\mathbf{Q}^{(p)}\right\| \leqslant q$ и $\left\|\mathbf{T}^{(p)}\right\|^2 \leqslant w_{\mathbf{T}}^2$.
    Тогда справедлива оценка
    \[
        \left\|\mathbf{H}_O\right\| \leqslant \sqrt{2}q^2\left\|\mathbf{x}\right\|^2(L+1)w_{\mathbf{T}}^{2L}.
    \]
\end{rustheorem}

\vspace{0.15em}
\key{Ключевой результат.} Единая факторизация $\mathbf{H}_O=\mathbf{Q}^{\top}\mathbf{F}^{\top}\mathbf{A}\mathbf{F}\mathbf{Q}$ переносит оценку нормы Гессе на произвольную суперпозицию линейных операторов.
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Схема доказательства: факторизация матрицы Гессе}
\justifying
\small
\begin{itemize}
    \item Цепное правило:
    \[
        \dfrac{\partial\mathbf{z}}{\partial\mathbf{W}^{(p)}} = \left(\mathbf{G}^{(p)} \otimes \mathbf{x}^{(p-1)^T}\right)\mathbf{Q}^{(p)}.
    \]
    \item Ключевое тождество векторизации $vec(\mathbf{B}\mathbf{V}\mathbf{A}^T) = (\mathbf{A} \otimes \mathbf{B})\,vec(\mathbf{V})$ порождает произведения Кронекера.
    \item Сборка блоков дает $\mathbf{H}_O^{(kl)} = \mathbf{Q}^{(k)^T}\mathbf{F}^{(k)}\mathbf{A}\mathbf{F}^{(l)^T}\mathbf{Q}^{(l)}$, откуда
    \[
        \mathbf{H}_O = \mathbf{Q}^{T}\mathbf{F}^{T}\mathbf{A}\mathbf{F}\mathbf{Q}.
    \]
    \item Матрицы $\mathbf{Q}$, $\mathbf{F}$ отражают архитектуру и параметризацию слоев, $\mathbf{A}$~--- внутренний гессиан softmax.
\end{itemize}

\vspace{0.15em}
\key{Ключевая идея.} Тождество векторизации превращает сборку блоков Гессе в произведение Кронекера $\mathbf{Q}$, $\mathbf{F}$, $\mathbf{A}$.
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Оценки нормы Гессе для трансформеров}
\justifying
\small

Матрица Гессе полного блока трансформера представляется по компонентам:
\begin{align*}
    \mathbf{H}_{\mathrm{tr}}^{(i,j)} &= \left( \mathbf{J}_Z \otimes \mathbf{I}_{n_i} \right) \boldsymbol{\xi}_{ij} \\
    &\quad + \left( \mathbf{I}_{N d_V} \otimes \mathbf{B}_i^\top \right) \mathbf{H}_Z \mathbf{B}_j,
\end{align*}
где $\mathbf{B}_i = \partial \mathbf{S} / \partial \mathbf{W}_i$, $\boldsymbol{\xi}_{ij} = \partial (\partial \mathbf{S} / \partial \mathbf{W}_i) / \partial \mathbf{W}_j$, а $\mathbf{J}_Z$ и $\mathbf{H}_Z$ обозначают матрицы Якоби и Гессе LayerNorm.

\begin{rustheorem}[\emph{Оценка нормы матрицы Гессе для слоя самовнимания}]
    \justifying
    Для слоя самовнимания $\mathbf{F}(\mathbf{X}) = \mathbf{A}(\mathbf{X})\mathbf{X}\mathbf{W}_V$ справедлива оценка $\left\|\mathbf{H}_i(\mathbf{w}^*)\right\|_2 \leqslant M$, где константа $M$ выражается через нормы $\mathbf{X}$, $\mathbf{W}_Q$, $\mathbf{W}_K$, $\mathbf{W}_V$, целевой переменной и размерности $N$, $d_V$, $d_K$.
\end{rustheorem}

\vspace{0.15em}
\key{Ключевой результат.} Для слоя самовнимания норма Гессе равномерно ограничена константой $M$, что дает основу оценки сходимости ландшафта трансформера.
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Оценки сходимости для блоков трансформера}
\justifying
\small
\begin{rustheorem}[\emph{Сходимость ландшафта функции потерь для трансформеров}]
    \justifying
    Для каждой пары~$(i,j)$ блоков трансформера справедлива оценка нормы
    \begin{align*}
        \left\|\mathbf{H}_{\mathrm{tr}}^{(i,j)}\right\|_2 &\leqslant \|\mathbf{J}_Z\|_2\,\|\boldsymbol{\xi}_{ij}\|_2 \\
        &\quad + \|\mathbf{B}_i\|_2\,\|\mathbf{H}_Z\|_2\,\|\mathbf{B}_j\|_2,
    \end{align*}
    а ландшафтная мера для одного блока самовнимания и блока трансформера подчиняется оценке сходимости
    \[
        \left| \mathcal{L}_{k+1}(\mathbf{w}) - \mathcal{L}_k(\mathbf{w}) \right| \leqslant \frac{2B_{\ell}}{k+1} + \frac{M \left\| \mathbf{w} - \mathbf{w}^* \right\|_2^2}{k+1},
    \]
    где константа $M$ вычисляется из оценок норм компонент, а $B_{\ell}$ обозначает границу функции потерь.
\end{rustheorem}

\begin{rusremark}
    \justifying
    В численных экспериментах наибольшие нормы наблюдаются у блоков Keys и Values.
\end{rusremark}

\vspace{0.15em}
\key{Ключевой результат.} Разность потерь убывает как $O(1/k)$, а скорость убывания определяется нормой Гессе блока трансформера.
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Схема доказательства: сходимость ландшафта для трансформеров}
\justifying
\small
\begin{enumerate}
    \item Разложение разности потерь на два слагаемых: вклад нового объекта и разность гессианов:
    \[
        \begin{aligned}
        \left|\mathcal{L}_{k+1}-\mathcal{L}_k\right| &\leqslant \underbrace{\frac{1}{k+1}\left|\ell(\mathbf{f}_{\mathbf{w}^*}(\mathbf{x}_{k+1}),\mathbf{y}_{k+1})-\frac{1}{k}\sum_{i=1}^k\ell(\mathbf{f}_{\mathbf{w}^*}(\mathbf{x}_i),\mathbf{y}_i)\right|}_{I} \\
        &\quad + \underbrace{\frac{\|\mathbf{w}-\mathbf{w}^*\|_2^2}{2(k+1)}\left\|\mathbf{H}_{k+1}-\frac{1}{k}\sum_{i=1}^k\mathbf{H}_i\right\|_2}_{II}.
        \end{aligned}
    \]
    \item Слагаемое $I$: из ограниченности потерь $0\leqslant\ell\leqslant B_\ell$ по неравенству треугольника получаем $\frac{2B_\ell}{k+1}$.
    \item Слагаемое $II$: из $\|\mathbf{H}_i(\mathbf{w}^*)\|_2\leqslant M$ по неравенству треугольника $\left\|\mathbf{H}_{k+1}-\frac{1}{k}\sum_i\mathbf{H}_i\right\|_2 \leqslant 2M$.
    \item Сборка: $\frac{2B_\ell}{k+1}+\frac{M\|\mathbf{w}-\mathbf{w}^*\|_2^2}{k+1}$; для блока самовнимания $M$ берется из оценки нормы Гессе.
\end{enumerate}

\vspace{0.15em}
\key{Ключевая идея.} Оба слагаемых убывают как $1/k$, а скорость определяется нормой Гессе $M$.
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Сверточные модели глубокого обучения}
\justifying
\footnotesize
\begin{rustheorem}[\emph{Оценка нормы матрицы Гессе для 1D-сверточных сетей}]
    \justifying
    Пусть задана функция нейронной сети~$f_{\boldsymbol{\theta}}(x) = C_{\mathbf{W}^{(L+1)}} \circ \sigma \circ \dots \circ \sigma \circ C_{\mathbf{W}^{(1)}},$
    где~$C_{\mathbf{W}^{(i)}}$~--- одномерная свертка с ядром~$\mathbf{W}^{(i)}$ без дополнения и с единичным шагом.
    Пусть заданы верхние оценки на параметры: $C_l \leqslant C$, $k_i \leqslant k_{\mathrm{ker}}$, $d_i \leqslant d_1 := d$, $|\mathbf{W}^{(p)}_{i,j,k}|^2 \leqslant w^2$.
    Тогда норма матрицы Гессе имеет верхнюю оценку
    \[
        \left\|\mathbf{H}_{O}\right\| \leqslant \sqrt{2}\left\|x\right\|^2d^2(L+1)(C^2w^2k_{\mathrm{ker}}d)^L.
    \]
\end{rustheorem}

\begin{rustheorem}[\emph{Оценка нормы матрицы Гессе для 2D-сверточных сетей}]
    \justifying
    Пусть задана функция нейронной сети~$f_{\boldsymbol{\theta}}(\mathbf{x}) = C_{\mathbf{W}^{(L+1)}} \circ \dots \circ C_{\mathbf{W}^{(1)}},$
    где $C_{\mathbf{W}^{(l)}}$~--- двумерная свертка с ядром $\mathbf{W}^{(i)}$ без дополнения и с единичным шагом.
    Пусть заданы верхние оценки на параметры: $C_l \leqslant C$, $k_i \leqslant k_{\mathrm{ker}}$, $m_i \leqslant m_1 := m$, $n_i \leqslant n_1 := n$, $|\mathbf{W}^{(p)}_{i,j,k}|^2 \leqslant w^2$.
    Тогда норма матрицы Гессе имеет верхнюю оценку
    \[
        \left\|\mathbf{H}_{O}\right\| \leqslant \sqrt{2}\left\|\mathbf{x}\right\|^2C^2k_{\mathrm{ker}}^2mn(L+1)(C^2k_{\mathrm{ker}}^2w^2mn)^{L}.
    \]
\end{rustheorem}

\begin{rusremark}
    \justifying
    Полученные оценки не учитывают уменьшение размеров после сверточных операций и зависят только от верхних границ параметров.
\end{rusremark}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Схема доказательства: оценка нормы Гессе для сверточных сетей}
\justifying
\footnotesize
\begin{itemize}
    \item Свертка как линейный оператор: векторизация дает $\mathbf{z}^{(p+1)} = \mathbf{T}^{(p+1)}\mathbf{x}^{(p)}$ с матрицей Теплица $\mathbf{T}^{(p)}$.
    \item Оценка нормы оператора: $\|\mathbf{T}^{(p)}\|^2 \leqslant C^2 w^2 k_{\mathrm{ker}} d$ (в 1D) и $\leqslant C^2 k_{\mathrm{ker}}^2 w^2 mn$ (в 2D, через структуру Теплица и ограничения $C_l\leqslant C$, $k_i\leqslant k_{\mathrm{ker}}$, $|W|^2\leqslant w^2$).
    \item Факторизация Гессе $\mathbf{H}_O = \mathbf{Q}^{\top}\mathbf{F}^{\top}\mathbf{A}\mathbf{F}\mathbf{Q}$ переносит ту же оценку на произведение $L+1$ множителей.
    \item Спектральная норма произведения: множитель $w_{\mathbf{T}}^{2L}$ от $L$ слоев; аддитивный член $(L+1)$; в 1D предмножитель $d^2$ от размера последовательности.
\end{itemize}

\vspace{0.15em}
\key{Ключевая идея.} Свертка задается матрицей Теплица, поэтому норма Гессе оценивается через произведение $L+1$ множителей.
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Асимптотика нормы Гессе для сверточных сетей}
\justifying

\begin{block}{Сводка асимптотик нормы $\|\mathbf{H}\|_2$}
\begin{tabular}{|l|c|}
\hline
\textbf{Архитектура} & \textbf{Асимптотика} \\
\hline
Полносвязная сеть & $L(hM)^{2L}$ \\
\hline
1D свертка & $L(C^2w^2k_{\mathrm{ker}}d)^L$ \\
\hline
2D свертка & $L(C^2k_{\mathrm{ker}}^2w^2mn)^L$ \\
\hline
С пулингом & $L\cdot\left(\frac{1}{k_{\mathrm{pool}}^2}\right)^{L-l+2}\cdot(\text{2D-базовая оценка})$ \\
\hline
\end{tabular}
\end{block}

\begin{itemize}
    \item Экспоненциальный рост с глубиной сети $L$; полиномиальный рост с размером слоев.
    \item Пулинг снижает сложность; оценки являются верхними границами и не зависят от процесса обучения модели.
\end{itemize}

\vspace{0.15em}
\key{Ключевой результат.} Все архитектуры подчиняются единой форме $L\cdot a^{L}$. Сложность ландшафта растет экспоненциально с глубиной.
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Сходимость ландшафта для сверточных сетей}
\justifying
\footnotesize
Пусть параметры $\boldsymbol{\theta}$ находятся в $R$-окрестности оптимума $\|\boldsymbol{\theta}-\boldsymbol{\theta}^*\|\le R$, функция потерь ограничена $|\ell_i|\le M_{\ell}$ и объекты ограничены $\|\mathbf{x}_i\|\le M_{\mathbf{x}}$.

\begin{rustheorem}[\emph{Сходимость ландшафта функции потерь для 1D-сверточных сетей}]
    \justifying
    При ограничениях теоремы об оценке нормы матрицы Гессе для 1D-сверточных сетей
    \[
        |\mathcal{L}_{k+1}(\boldsymbol{\theta})-\mathcal{L}_k(\boldsymbol{\theta})|\le\frac{2}{k+1}M_{\ell}+\frac{2}{k+1}R^2\sqrt{2}d^2M_{\mathbf{x}}^2(L+1)(C^2w^2k_{\mathrm{ker}}d)^L.
    \]
\end{rustheorem}

\begin{rustheorem}[\emph{Сходимость ландшафта функции потерь для 2D-сверточных сетей}]
    \justifying
    При ограничениях теоремы об оценке нормы матрицы Гессе для 2D-сверточных сетей
    \[
        |\mathcal{L}_{k+1}(\boldsymbol{\theta})-\mathcal{L}_k(\boldsymbol{\theta})|\le\frac{2}{k+1}M_{\ell}+\frac{2}{k+1}R^2\sqrt{2}q^2M_{\mathbf{x}}^2(L+1)(C^2k_{\mathrm{ker}}^2w^2mn)^L.
    \]
\end{rustheorem}

Здесь $C$~--- максимальное число каналов, $d$~--- длина входной последовательности, $m,n$~--- размеры входного изображения, $k_{\mathrm{ker}}$~--- размер свертки, $L$~--- глубина сети, $q^2=C^2k_{\mathrm{ker}}^2mn$.

\begin{ruscorollary}[\emph{Асимптотика ландшафтной меры}]
    \justifying
    Для 1D-сверток $\mu_f(f|D)\propto L(C^2M^2k_{\mathrm{ker}}d)^{L}$; для 2D-сверток $\mu_f(f|D)\propto C^2k_{\mathrm{ker}}^2L(C^2k_{\mathrm{ker}}^2M^2mn)^{L}$. Здесь $M$~--- некоторая константа.
\end{ruscorollary}
\end{frame}

\section{Ландшафтные меры для некоторых семейств}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Ландшафтная мера полносвязной сети}
\justifying
\small

\begin{rustheorem}[\emph{Сходимость ландшафта функции потерь для полносвязных сетей}]
Пусть параметры $\boldsymbol{\theta}$ выбраны так, что $\left\|\boldsymbol{\theta} - \boldsymbol{\theta}^*\right\|_2^2 \leqslant R^2$ для некоторого $R > 0$. Если существует неотрицательная константа $M_{\ell}$ такая, что $\left| \ell(f_{\boldsymbol{\theta}^*}(\mathbf{x}_{i}), \mathbf{y}_{i}) \right| \leqslant M_{\ell}$ для всех объектов $i = 1, \ldots, m$, то при выполнении условий теоремы о норме матрицы Гессе для полносвязных сетей справедливо:
\begin{align*}
\left|\mathcal{L}_{k+1}(\boldsymbol{\theta}) - \mathcal{L}_k(\boldsymbol{\theta})\right| &\leqslant \frac{1}{k+1}\Biggl( 2M_{\ell} \\
&\quad + \Biggl( L \sqrt{2} M_{\mathbf{x}}^2 M_{\mathbf{W}}^{2L} + \sqrt{2} \frac{M_{\mathbf{W}}^2 (M_{\mathbf{W}}^{2L} - 1)}{M_{\mathbf{W}}^2 - 1} \Biggr) R^2 \Biggr),
\end{align*}
причем выражение асимптотически стремится к нулю при~$k \to \infty$.
\end{rustheorem}

\begin{ruscorollary}
Ландшафтная мера сложности полносвязной модели: $\mu_f(f|D) \propto L(hM)^{2L}.$
\end{ruscorollary}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Интерпретация ландшафтной меры}
\justifying

\textbf{Интерпретация:} сложность ландшафта растет \textbf{экспоненциально} с глубиной $L$ и \textbf{полиномиально} с шириной $h$. Это объясняет, почему глубокие сети требуют больше данных для обучения.

\begin{rusremark}
    \justifying
    Для однослойной сети ($L=1$) имеем $\mu_f \propto h^2$~--- квадратичная зависимость от ширины.
\end{rusremark}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Локальная оценка ландшафтного приращения для квадратичной потери}
\justifying
\footnotesize
В регрессионной постановке с квадратичной функцией потерь $\ell_i(\boldsymbol{\theta})=\frac{1}{2}(f_{\boldsymbol{\theta}}(\mathbf{x}_i)-y_i)^2$ и полносвязными ReLU-сетями без смещений при ограничениях $\|\mathbf{W}^{(p)}\|_2\le M_{\mathbf{W}}$, $\|\mathbf{x}_i\|_2\le M_{\mathbf{x}}$ существует константа $M_{\mathbf{H}}:=L M_{\mathbf{x}}^2 M_{\mathbf{W}}^{2L}+\sum_{s=1}^{L}M_{\mathbf{W}}^{2s}$, такая, что $\|\mathbf{H}_i(\boldsymbol{\theta})\|_2\le M_{\mathbf{H}}$ для каждого $i$. При покоординатном ограничении $|w_{ij}^{(p)}|\le M$ имеем $\|\mathbf{H}_i(\boldsymbol{\theta})\|_2=O(L(hM)^{2L})$.

\begin{rustheorem}[\emph{Локальная оценка ландшафтного приращения}]
    \justifying
    Пусть $\|\boldsymbol{\theta}-\boldsymbol{\theta}^*\|_2\le R$, $|\ell(f_{\boldsymbol{\theta}^*}(\mathbf{x}_i),\mathbf{y}_i)|\le M_{\ell}$ для всех $i$ и выполнена приближенная стационарность с некоторым $\varepsilon_g\ge0$. Тогда
    \[
        |\mathcal{L}_{k+1}(\boldsymbol{\theta})-\mathcal{L}_k(\boldsymbol{\theta})|\le\frac{2}{k+1}(M_{\ell}+M_{\mathbf{H}}R^2)+2\varepsilon_g R,
    \]
    вклад кривизны имеет порядок $O(L(hM)^{2L}R^2/k)$.
\end{rustheorem}

\begin{ruscorollary}[\emph{Линейная модель}]
    \justifying
    Если $f_{\boldsymbol{\theta}}(\mathbf{x})=\mathbf{w}^{\top}\mathbf{x}$, то $\mathbf{H}_i(\boldsymbol{\theta})=\mathbf{x}_i\mathbf{x}_i^{\top}$ не зависит от $\mathbf{w}$ и $y_i$, поэтому $\mu_f(f|D)=O(n)$ при $\|\mathbf{x}_i\|_2^2=\Theta(n)$.
\end{ruscorollary}

\begin{ruscorollary}[\emph{Один скрытый слой, $L=2$}]
    \justifying
    Худший случай: $\mu_f(f|D)=O((hM)^4)$; при малых остатках $|f_{\boldsymbol{\theta}^*}(\mathbf{x}_i)-y_i|\le\varepsilon_y$ и $n\asymp h$ ведущий член имеет порядок $O(M^2h^3)$, что мягче $O(h^4)$.
\end{ruscorollary}
\end{frame}

\section{Достаточный размер выборки}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Сэмплирование эмпирической функции ошибки}
\justifying
Критерий согласованности на простой совокупности сводится к минимальному размеру выборки~$m^*$. Получим его вычислимую оценку.

\begin{rusdefinition}
    \justifying
    Пусть задано некоторое~$\varepsilon > 0$.
    Размер выборки~$m^*$ назовем D-достаточным, если для всех~$k\geqslant m^*$ выполняется условие:~$D(k) = \mathbb{D}_{\hat{\mathbf{w}}_{k}} L(\mathfrak{D}_m, \hat{\mathbf{w}}_{k}) \leqslant \varepsilon.$
\end{rusdefinition}

\begin{rusdefinition}
    \justifying
    Пусть задано некоторое~$\varepsilon > 0$.
    Размер выборки~$m^*$ назовем M-достаточным, если для всех~$k\geqslant m^*$ выполняется условие:~$M(k) = \left| \mathbb{E}_{\hat{\mathbf{w}}_{k+1}} L(\mathfrak{D}_m, \hat{\mathbf{w}}_{k+1}) - \mathbb{E}_{\hat{\mathbf{w}}_{k}} L(\mathfrak{D}_m, \hat{\mathbf{w}}_{k}) \right| \leqslant \varepsilon.$
\end{rusdefinition}

\begin{rustheorem}[\emph{Корректность M-достаточного размера выборки}]
    \justifying
    Пусть~$\|\mathbf{m}_{k+1} - \mathbf{m}_k\|_2 \to 0$ и~$\|\bm{\Sigma}_{k+1} - \bm{\Sigma}_k\|_{F}\to 0$ при~$k\to \infty$. 
    Тогда в модели линейной регрессии определение M-достаточного размера выборки корректно.
    А именно, для любого~$\varepsilon > 0$ существует такой~$m^*$, что для всех~$k\geqslant m^*$ выполняется~$M(k)\leqslant\varepsilon$.
\end{rustheorem}

\begin{ruscorollary}
    \justifying
    Пусть~$\|\mathbf{m}_k - \mathbf{w}\|_2\to 0$ и~$\|\bm{\Sigma}_k - \left[k\mathcal{I}(\mathbf{w})\right]^{-1}\|_{F}\to 0$ при~$k \to \infty$.
    Тогда в модели линейной регрессии определение M-достаточного размера выборки корректно.
\end{ruscorollary}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Схема доказательства: корректность M-достаточности}
\justifying
\small
\begin{enumerate}
    \item Асимптотическая нормальность оценки максимального правдоподобия: $\sqrt{k}(\hat{\mathbf{w}}_k-\mathbf{w})\xrightarrow{d}\mathcal{N}\left(0,\mathcal{I}^{-1}(\mathbf{w})\right)$; сходимость моментов в линейной регрессии устанавливается теоремой о сходимости апостериорных распределений.
    \item Переход к логарифму правдоподобия $l(\mathfrak{D}_m,\hat{\mathbf{w}}_k)$: произведение заменяется суммой, что упрощает анализ разности.
    \item Непрерывность: при сходимости моментов $\mathbf{m}_{k+1}\to\mathbf{m}_k$ и $\bm{\Sigma}_{k+1}\to\bm{\Sigma}_k$ отображение $\hat{\mathbf{w}}\mapsto l(\mathfrak{D}_m,\hat{\mathbf{w}})$ непрерывно и равномерно ограничено.
    \item Следствие: $M(k) = \left|\mathbb{E}\,l(\mathfrak{D}_m,\hat{\mathbf{w}}_{k+1})-\mathbb{E}\,l(\mathfrak{D}_m,\hat{\mathbf{w}}_k)\right|\to 0$, откуда для любого $\varepsilon>0$ существует $m^*$ с $M(k)\leqslant\varepsilon$ при всех $k\geqslant m^*$.
\end{enumerate}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{KL- и S-близость апостериорных распределений}
\justifying
\small
\begin{rusdefinition}
    \justifying
    Подвыборки~$\mathfrak{D}^1$ и~$\mathfrak{D}^2$ назовем близкими, если~$\mathcal{I}_2$ может быть получено из~$\mathcal{I}_1$ путем удаления, замены или добавления одного элемента, то есть
    \[
        \left| \mathcal{I}_1 \triangle \mathcal{I}_2 \right| = \left| \left( \mathcal{I}_1 \setminus \mathcal{I}_2 \right) \cup \left( \mathcal{I}_2 \setminus \mathcal{I}_1 \right) \right| = 1.
    \]
\end{rusdefinition}

\begin{rusdefinition}
    \justifying
    Пусть задано некоторое~$\varepsilon > 0$.
    Размер выборки~$m^*$ называется KL-достаточным, если для всех~$k\geqslant m^*$
    \[
        KL(k) = D_{KL}(p_k \| p_{k+1}) = \int p_k(\mathbf{w}) \log{\frac{p_k(\mathbf{w})}{p_{k+1}(\mathbf{w})}} d\mathbf{w} \leqslant \varepsilon.
    \]
    Размер выборки~$m^*$ называется S-достаточным, если для всех~$k\geqslant m^*$
    \[
        S(k) = \text{s-score}(p_k, p_{k+1}) \geqslant 1-\varepsilon.
    \]
\end{rusdefinition}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Теоремы сходимости KL- и S-критериев}
\justifying
\small
\begin{rustheorem}[\emph{Корректность KL- и S-достаточного размера выборки}]
    \justifying
    Пусть~$\|\mathbf{m}_{k+1} - \mathbf{m}_k\|_2 \to 0$ и~$\|\bm{\Sigma}_{k+1} - \bm{\Sigma}_k\|_{F}\to 0$ при~$k\to \infty$.
    Тогда в модели с нормальным апостериорным распределением параметров определения KL- и S-достаточного размера выборки корректны: для любого~$\varepsilon > 0$ существует~$m^*$ такое, что для всех~$k\geqslant m^*$ выполняется~$KL(k)\leqslant\varepsilon$ и~$S(k)\geqslant 1-\varepsilon$.
\end{rustheorem}

\begin{rustheorem}[\emph{Сходимость апостериорных распределений в линейной регрессии}]
    \justifying
    Пусть множества значений признаков и целевой переменной ограничены: $\|\mathbf{x}\|_2\leqslant M$ и~$|y|\leqslant M$ для всех объектов выборки.
    Если~$\lambda_{\min}\left(\mathbf{X}^{\top}_k \mathbf{X}_k \right) = \omega(\sqrt{k})$ при~$k\to \infty$, то в модели линейной регрессии с нормальным априорным распределением параметров~$\|\mathbf{m}_{k+1} - \mathbf{m}_k\|_2\to 0$ и~$\|\bm{\Sigma}_{k+1} - \bm{\Sigma}_k\|_{F}\to 0$ при~$k\to \infty$.
\end{rustheorem}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Схема доказательства: сходимость KL- и S-критериев}
\justifying
\small
\begin{enumerate}
    \item Сопряженность: апостериорное распределение нормальное, $\mathbf{\Sigma} = \left(\alpha\mathbf{I} + \frac{1}{\sigma^2}\mathbf{X}^{\top}\mathbf{X}\right)^{-1}$, $\mathbf{m} = \frac{1}{\sigma^2}\mathbf{\Sigma}\mathbf{X}^{\top}\mathbf{y}$.
    \item Разность ковариаций: тождество для разности обратных матриц дает $\|\mathbf{\Sigma}_{k+1}-\mathbf{\Sigma}_k\|_2 \leqslant \sigma^2 M^2 \cdot \dfrac{1}{\lambda_{\min}(\mathbf{X}_{k+1}^{\top}\mathbf{X}_{k+1})}\cdot\dfrac{1}{\lambda_{\min}(\mathbf{X}_k^{\top}\mathbf{X}_k)}$.
    \item Условие $\lambda_{\min}(\mathbf{X}_k^{\top}\mathbf{X}_k) = \omega(\sqrt{k})$ дает $1/\lambda_{\min} = o(1/\sqrt{k})$, откуда $\|\mathbf{\Sigma}_{k+1}-\mathbf{\Sigma}_k\|_2 = o(1/k)$, а по норме Фробениуса~--- тоже $o(1/k)$.
    \item Разность средних: лемма об обращении матриц и оценка $\|\mathbf{X}_k^{\top}\mathbf{y}_k\|_2 \leqslant kM^2$ дают $\|\mathbf{m}_{k+1}-\mathbf{m}_k\|_2 = o(1)$.
    \item Сходимость моментов влечет $KL(k)\to 0$ (по непрерывности KL) и $s\text{-score}\to 1$, что и дает корректность критериев.
\end{enumerate}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Эксперименты по KL- и S-достаточности}
\justifying
\small

\begin{figure}[h!t]\center
    \includegraphics[width=0.36\textwidth]{thesis/figures/chapter-3/posterior-distribution/dependence_on_available_sample_set}
    \includegraphics[width=0.36\textwidth]{thesis/figures/chapter-3/statical/graphs}
\end{figure}

\begin{itemize}
    \item Функции $KL(k)$ и $S(k)$ с ростом~$k$ асимптотически стремятся к константе.
    \item KL-достаточность дает консервативные оценки, а S-достаточность дает меньшие достаточные объемы.
    \item Результаты согласуются с теоремами о сходимости предложенных критериев.
\end{itemize}

\key{Ключевой результат.} Критерии KL- и S-достаточности дают вычислимые оценки минимального размера выборки, согласованные с теоремами сходимости.
\end{frame}

\section{Снижение сложности моделей}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Байесовская дистилляция: сохранение нормальности апостериорного распределения}
\justifying
\footnotesize
Переходим к методам снижения сложности моделей. Байесовская дистилляция передает знание учителя ученику через априорное распределение параметров. При совпадении архитектур учителя и ученика априорное распределение параметров ученика задается непосредственно апостериорным распределением параметров учителя~$p(\mathbf{u}|\mathfrak{D})=\mathcal{N}(\mathbf{m},\boldsymbol{\Sigma})$. При различающихся архитектурах требуется выравнивание пространств параметров.

\begin{rustheorem}[\emph{Распределение параметров при удалении нейрона}]
    \justifying
    Пусть число слоев модели учителя равно числу слоев модели ученика $T=L$; размеры соответствующих слоев удовлетворяют условию $n_t^{(\mathrm{tr})}\ge n_t^{(\mathrm{st})}$. Пусть преобразование $\boldsymbol{\phi}(t,\mathbf{u})$ удаляет один нейрон из $t$-го слоя модели учителя, а $\boldsymbol{\upsilon}=\boldsymbol{\phi}(t,\mathbf{u})$~--- вектор оставшихся параметров. Тогда распределение параметров после такого преобразования также является нормальным: $p(\boldsymbol{\upsilon}|\mathfrak{D})=\mathcal{N}(\boldsymbol{\mu}_{\boldsymbol{\upsilon}},\boldsymbol{\Xi}_{\boldsymbol{\upsilon},\boldsymbol{\upsilon}})$ с параметрами условного нормального распределения.
\end{rustheorem}

\begin{rustheorem}[\emph{Распределение параметров при удалении слоя}]
    \justifying
    Пусть размеры соседних слоев модели учителя совпадают $n_t^{(\mathrm{tr})}=n_{t-1}^{(\mathrm{tr})}$, то есть матрица $\mathbf{U}_t$ является квадратной; функция активации удовлетворяет свойству идемпотентности $\boldsymbol{\sigma}\circ\boldsymbol{\sigma}=\boldsymbol{\sigma}$. Пусть преобразование $\boldsymbol{\psi}(t,\mathbf{u})$ удаляет $t$-й слой модели учителя, а $\boldsymbol{\upsilon}=\boldsymbol{\psi}(t,\mathbf{u})$~--- вектор оставшихся параметров. Тогда апостериорное распределение оставшихся параметров также является нормальным.
\end{rustheorem}

\begin{rusremark}
    \justifying
    Сохранение нормальности выравнивает пространства параметров учителя и ученика при различающихся архитектурах.
\end{rusremark}
\end{frame}

\begin{frame}[t]{Многодоменная дистилляция}
\justifying
\small

\begin{figure}[h!t]\center
    {\includegraphics[width=0.32 \textwidth]{thesis/figures/chapter-4/multidomain-distillation/Distillation_only}}
    {\includegraphics[width=0.32 \textwidth]{thesis/figures/chapter-4/multidomain-distillation/Distillation_DA}}
\end{figure}

\begin{rusdefinition}
    \justifying
    Генеральная совокупность~$B$ называется близкой к генеральной совокупности~$A$, если существует инъективное отображение $\varphi: A \rightarrow B$.
\end{rusdefinition}

Заданы выборки из двух близких генеральных совокупностей отображением~$\varphi$:
\[
    \mathfrak{D}_{A}=\{(\mathbf{x_i},\mathbf{y_i})\}_{i=1}^n,
\mathbf{x_i} \in \mathbb{X}_{A},
\mathbf{y_i} \in \mathbb{Y}, \qquad \mathfrak{D}_{B}=\{(\mathbf{x_i},\mathbf{y_i})\}_{i=1}^m, \mathbf {x_i} \in \mathbb{X}_{B}, \mathbf{y_i} \in \mathbb{Y}.
\]
Оптимизационная задача для многодоменной дистилляции:
\begin{align*}
    \mathcal{L}(\mathbf{w,X,Y,f,\varphi}) =&
    -\lambda\sum\limits_{i=1}^{m}\sum\limits_{r=1}^{ R}\mathbb{I}[y_{i}=r]\log{g^{r}(\mathbf{x}_{i},\mathbf{w})} -\\
    &-(1-\lambda)\sum\limits_{i=1}^{m}\sum\limits_{r=1}^{R}(f\circ \varphi)^{r}(\mathbf{x }_{i})\log{g^{r}(\mathbf{x}_{i},\mathbf{w})}.
\end{align*}

\key{Ключевая идея.} Учитель и ученик работают на близких генеральных совокупностях, а знание передается через отображение $\varphi$.
\end{frame}

\begin{frame}{Антидистилляция: передача знаний от простой модели к сложной}
\justifying
\small

\textbf{Идея:} модель-учитель ($g_{\text{tr}}$) обучена на простой задаче $\mathfrak{D}_1$; нужно обучить более сложную модель-ученика ($f_{\text{st}}$) на сложной задаче $\mathfrak{D}_2$, используя знания учителя.

\textbf{Проблема:} размерности моделей разные ($N_{\text{tr}} < N_{\text{st}}$). Решение~--- составная функция потерь при инициализации:

\[
\mathcal{L}(\mathbf{w}) = \lambda_1\mathcal{L}_{\text{ce}}(\mathbf{w},\mathfrak{D}_1) + \lambda_2\|\mathbf{u}-\mathbf{Pr}[\mathbf{w}]\|^2 + \lambda_3\mathcal{L}_3^{\delta}(\mathbf{w},\mathfrak{D}_1) + \lambda_4\operatorname{tr}(\nabla^2\mathcal{L}_{\text{ce}}).

Здесь слагаемое с $\lambda_2$ сближает параметры учителя и ученика через проекцию $\mathbf{Pr}$, а слагаемое с $\lambda_4$ регуляризует след матрицы Гессе функции потерь.

\begin{frame}{Теоремы о LoRA-адаптерах}
\justifying
\footnotesize
LoRA-адаптеры снижают число обучаемых параметров при тонкой настройке. Докажем, что состоятельность оценки при этом сохраняется. Для адаптации предобученного классификатора рассматривается низкоранговое обновление
\[
    \Delta W = AB,\quad A\in\mathbb{R}^{d\times r},\ B\in\mathbb{R}^{r\times k},\ r\ll\min(d,k),
\]
число обучаемых параметров~$r(d+k)$ вместо~$dk$.

\begin{rustheorem}[\emph{Состоятельность оценки}]
    \justifying
    Пусть $\mathcal{D}=\{(X_i,c_i)\}_{i=1}^n$~--- независимые одинаково распределенные пары из распределения $P_{\mathrm{true}}$ на $\mathcal{X}\times[N_c]$, где $[N_c]=\{1,\ldots,N_c\}$. Рассмотрим компактное множество $K\subset\mathbb{R}^p$ и параметрическую модель $P_{\Phi_0+\Theta}(c\mid X)$ при $\Theta\in K$, где $\Phi_0$~--- замороженный вектор предобученных параметров. Предположим: существует единственная точка $\Theta^\star\in K$, минимизирующая ожидаемый риск $L(\Theta)=\mathbb{E}_{(X,c)\sim P_{\mathrm{true}}}[\log P_{\Phi_0+\Theta}(c\mid X)]$; функция $(X,c,\Theta)\mapsto-\log P_{\Phi_0+\Theta}(c\mid X)$ непрерывна и $\mathbb{E}[\sup_{\Theta\in K}|\log P_{\Phi_0+\Theta}(c\mid X)|]<\infty$; эмпирический риск задан в виде $\mathcal{L}_n(\Theta)=-\frac{1}{n}\sum_{i=1}^n\log P_{\Phi_0+\Theta}(c_i\mid X_i)$, а оценка $\widehat{\Theta}_n$ выбирается как измеримый минимизатор $\mathcal{L}_n$ на $K$. Тогда $\widehat{\Theta}_n\xrightarrow{P}\Theta^\star$ при $n\to\infty$.
\end{rustheorem}

\begin{rustheorem}[\emph{Корректность при низкоранговых обновлениях}]
    \justifying
    Пусть выходной слой задается в виде $\hat{\mathbf{y}}=\operatorname{softmax}(W_{\mathrm{upd}}^{\top}\mathbf{x})$, где $W\in\mathbb{R}^{d\times k}$~--- замороженные предобученные веса, а $W_{\mathrm{upd}}=W+\Delta W$. Вместо непосредственного обучения $\Delta W\in\mathbb{R}^{d\times k}$ обновление параметризуется в низкоранговой форме $\Delta W=AB$, $A\in\mathbb{R}^{d\times r}$, $B\in\mathbb{R}^{r\times k}$, $r\ll\min(d,k)$. Пусть выполнены условия теоремы о состоятельности оценки. Тогда выход модели $\hat{\mathbf{y}}$ сохраняется, то есть низкоранговое обновление не искажает корректность классификационного слоя.
\end{rustheorem}
\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Следствие о LoRA}
\justifying
\footnotesize
\begin{ruscorollary}[\emph{LoRA с теоретическим обоснованием}]
    \justifying
    Тонкая настройка на основе низкоранговой адаптации одновременно сохраняет статистическую состоятельность оценки и корректность выходного слоя, при этом требуя существенно меньшего числа обучаемых параметров.
\end{ruscorollary}

\begin{itemize}
    \item Объединяя теорему о состоятельности оценки и теорему о корректности при низкоранговых обновлениях, получаем, что низкоранговая адаптация не ухудшает статистическую состоятельность и не искажает классификационный слой.
    \item В экспериментах LoRA с рангом $r=8$ требует только $0{,}16\%$ параметров полной модели DeBERTa-v3-base, а снижение метрик качества относительно полной тонкой настройки составляет порядка $1{,}4$--$3{,}2\%$.
    \item Уменьшение числа обучаемых параметров снижает сложность модели, сохраняя качество аппроксимации, что согласуется с пятым положением, выносимым на защиту.
\end{itemize}
\end{frame}

\begin{frame}{Радемахеровская сложность многозадачного обучения}
\justifying
\footnotesize
LoRA снижает сложность модели параметризацией. Оценим, как снижается сложность на задачу при многозадачном обучении.

\begin{rustheorem}[\emph{Сложность Радемахера на задачу при MTL}]
    \justifying
    Пусть $S_t=\{x_i\}_{i=1}^n$~--- выборка фиксированной целевой задачи $t$ с $\|x_i\|_2 \le R$.
    Пусть $\phi(\cdot;w)$~--- энкодер, и рассмотрим линейные головы
    $f_{w,h}(x)=h^\top \phi(x;w)$ с $\|h\|_2 \le B_{\mathrm{head}}$.
    Предположим:
    \begin{enumerate}
        \item $\|\phi(x;w)\|_2 \le B_{\mathrm{Lip}}\,\|w\|_2\,\|x\|_2$ для всех $x,w$.
        \item Энкодер STL удовлетворяет $\|w_{\mathrm{enc}}\|\le B_{\mathrm{enc}}$, общий энкодер MTL удовлетворяет $\|w_{\mathrm{shared}}\|\le B_{\mathrm{shared}}$.
        \item Многозадачное масштабирование: $B_{\mathrm{shared}} \le B_{\mathrm{enc}}/\sqrt{T}$.
    \end{enumerate}

Пусть $\mathcal{F}_{\mathrm{STL}}^{(t)}$ совпадает с семейством $\mathcal{F}(B_{\mathrm{enc}})$, а $\mathcal{F}_{\mathrm{MTL}}^{(t)}$~--- с семейством $\mathcal{F}(B_{\mathrm{shared}})$, и пусть $U_n(B) := \frac{B_{\mathrm{head}}B_{\mathrm{Lip}}RB}{\sqrt{n}}$.

    Тогда справедливы оценки сложности:
    \[
        \widehat{\mathfrak R}_n\!\left(\mathcal{F}_{\mathrm{MTL}}^{(t)};S_t\right)\le U_n(B_{\mathrm{shared}})\le \frac{1}{\sqrt{T}}\,U_n(B_{\mathrm{enc}}),
        \qquad
        \widehat{\mathfrak R}_n\!\left(\mathcal{F}_{\mathrm{STL}}^{(t)};S_t\right)\le U_n(B_{\mathrm{enc}}).
    \]
\end{rustheorem}

\begin{rusremark}
    \justifying
    Данный анализ сохраняет размер выборки $n$ фиксированным и сравнивает STL и MTL при одном~$n$: улучшение проявляется множителем $1/\sqrt{T}$ в радемахеровской сложности на задачу.
\end{rusremark}

\end{frame}
%----------------------------------------------------------------------------------------------------------
\begin{frame}{Схема доказательства: сложность Радемахера при MTL}
\justifying
\footnotesize

\textbf{Шаг 1. Класс гипотез.} Для $B>0$
\[
    \mathcal{F}(B) = \bigl\{x \mapsto h^{\top}\phi(x;w) \bigm| \|h\|_2 \le B_{\mathrm{head}},\ \|w\|_2 \le B\bigr\}.
\]

\textbf{Шаг 2. Дуальность Коши--Буняковского.} Для фиксированных $w$ и $\sigma$
\[
    \sup_{\|h\|_2 \le B_{\mathrm{head}}}\sum_{i=1}^{n}\sigma_i\, h^{\top}\phi(x_i;w) = B_{\mathrm{head}}\Bigl\|\sum_{i=1}^{n}\sigma_i\, \phi(x_i;w)\Bigr\|_2.
\]

\textbf{Шаг 3. Неравенства Йенсена и Хинчина.}
\[
    \mathbb{E}_{\sigma}\Bigl\|\sum_{i=1}^{n}\sigma_i a_i\Bigr\|_2 \le \Bigl(\sum_{i=1}^{n}\|a_i\|_2^2\Bigr)^{1/2}.
\]

\textbf{Шаг 4. Оценка.} С $\|\phi(x;w)\|_2 \le B_{\mathrm{Lip}}\|w\|_2\|x\|_2$ и $\|x_i\|_2 \le R$
\[
    \widehat{\mathfrak R}_n\bigl(\mathcal{F}(B);S_t\bigr) \le \frac{B_{\mathrm{head}}B_{\mathrm{Lip}}RB}{\sqrt{n}} = U_n(B).
\]

\textbf{Шаг 5. Множитель $1/\sqrt{T}$.} По предположению $B_{\mathrm{shared}} \le B_{\mathrm{enc}}/\sqrt{T}$
\[
    \widehat{\mathfrak R}_n\bigl(\mathcal{F}_{\mathrm{MTL}}^{(t)};S_t\bigr) \le U_n(B_{\mathrm{shared}}) \le \frac{1}{\sqrt{T}}\,U_n(B_{\mathrm{enc}}).
\]

\end{frame}
